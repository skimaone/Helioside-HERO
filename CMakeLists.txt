cmake_minimum_required(VERSION 3.22)
project(SuntechHero VERSION 0.1.4 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  add_compile_options(/utf-8 /fp:precise)
endif()
include(CTest)
add_subdirectory(research)
# The supplied regression programs use assert(): keep them active in Release.
get_property(research_targets DIRECTORY research PROPERTY BUILDSYSTEM_TARGETS)
foreach(t IN LISTS research_targets)
  get_target_property(kind ${t} TYPE)
  if(kind STREQUAL "EXECUTABLE")
    target_compile_options(${t} PRIVATE $<IF:$<CXX_COMPILER_ID:MSVC>,/UNDEBUG,-UNDEBUG>)
  endif()
endforeach()
add_executable(hero_core_tests tests/core_tests.cpp)
target_include_directories(hero_core_tests PRIVATE core)
add_test(NAME production_core COMMAND hero_core_tests)
add_executable(hero_colour_tests tests/colour_tests.cpp)
target_include_directories(hero_colour_tests PRIVATE core)
add_test(NAME colour_contract COMMAND hero_colour_tests)
option(HERO_BUILD_PLUGIN "Build JUCE VST3 and host tests" ON)
if(NOT HERO_BUILD_PLUGIN)
  return()
endif()
set(JUCE_DIR "" CACHE PATH "Local JUCE 8.0.4 checkout")
if(JUCE_DIR)
  add_subdirectory("${JUCE_DIR}" JUCE)
else()
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.4 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
endif()
file(GLOB HERO_IMAGES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/resources/gui/*.png")
juce_add_binary_data(hero_assets HEADER_NAME HeroAssets.h NAMESPACE HeroAssets SOURCES
  ${HERO_IMAGES} resources/gui/DejaVuSansMono.ttf resources/gui/DejaVu_LICENSE.txt)
juce_add_plugin(HERO COMPANY_NAME "Suntech Audio" PLUGIN_MANUFACTURER_CODE Sntc PLUGIN_CODE Hero
  PRODUCT_NAME "Suntech Audio HERO" BUNDLE_ID audio.suntech.hero
  FORMATS VST3 Standalone IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
  IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Dynamics)
target_sources(HERO PRIVATE plugin/Processor.cpp plugin/Editor.cpp)
target_include_directories(HERO PUBLIC plugin core "$<TARGET_PROPERTY:juce::juce_core,INTERFACE_INCLUDE_DIRECTORIES>")
target_compile_definitions(HERO PUBLIC JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
  JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(HERO PRIVATE hero_assets juce::juce_audio_utils juce::juce_dsp juce::juce_recommended_config_flags)
if(MSVC)
  target_compile_options(HERO PRIVATE $<$<CONFIG:Release>:/Zi>)
  target_link_options(HERO_VST3 PRIVATE $<$<CONFIG:Release>:/DEBUG:FULL> $<$<CONFIG:Release>:/MAP:${CMAKE_BINARY_DIR}/SuntechHero-Release.map>)
  set_target_properties(HERO_VST3 PROPERTIES PDB_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/symbols")
endif()
add_custom_command(TARGET HERO_VST3 POST_BUILD
  COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_FILE_DIR:HERO_VST3>/../Resources/Licenses"
  COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" "$<TARGET_FILE_DIR:HERO_VST3>/../Resources/Licenses/DSP_MIT.txt"
  COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/resources/gui/DejaVu_LICENSE.txt" "$<TARGET_FILE_DIR:HERO_VST3>/../Resources/Licenses/DejaVu_LICENSE.txt")
if(BUILD_TESTING)
  add_executable(hero_tests tests/plugin_tests.cpp)
  target_link_libraries(hero_tests PRIVATE HERO)
  add_test(NAME processor COMMAND hero_tests processor "${CMAKE_BINARY_DIR}/test-results/processor")
  add_test(NAME editor COMMAND hero_tests editor "${CMAKE_BINARY_DIR}/test-results/gui")
  add_test(NAME alias_and_cpu COMMAND hero_tests analysis "${CMAKE_BINARY_DIR}/test-results/analysis")
  add_executable(hero_vst3_host tests/vst3_host.cpp)
  target_link_libraries(hero_vst3_host PRIVATE HERO)
  add_test(NAME native_vst3 COMMAND hero_vst3_host "$<TARGET_FILE_DIR:HERO_VST3>/../.." "${CMAKE_BINARY_DIR}/test-results/native")
  set_tests_properties(processor editor alias_and_cpu native_vst3 PROPERTIES TIMEOUT 300)
endif()
